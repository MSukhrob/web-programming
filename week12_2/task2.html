Code Challenge Specification (Slide 31)

Build a GitHub User Lookup component that manages async network requests with useEffect, handles loading and 404 errors, and aborts in-flight requests when the username changes.

  • Username Input: Controlled input field with Enter key and Search button trigger.
  • Data Fetching: https://api.github.com/users/{username} executed within useEffect.
  • Data Display: Avatar, display name, bio, follower & following counts, public repository count, location, and GitHub profile link.
  • Bonus: Cancel prior in-flight fetch requests with controller.abort() on new username input.

// 07-GitHubUserLookup.jsx - Full Solution
import { useState, useEffect } from 'react';

export default function GitHubUserLookup() {
  const [input, setInput]       = useState('muxiddin19');
  const [username, setUsername] = useState('muxiddin19');
  const [user, setUser]         = useState(null);
  const [loading, setLoading]   = useState(false);
  const [error, setError]       = useState(null);

  useEffect(() => {
    if (!username) return;
    const controller = new AbortController(); // BONUS: abort previous request
    setLoading(true);
    setError(null);
    setUser(null);

    fetch(`https://api.github.com/users/${encodeURIComponent(username)}`, { signal: controller.signal })
      .then(res => {
        if (res.status === 404) throw new Error('User not found (404)');
        if (!res.ok) throw new Error('Request failed with HTTP status ' + res.status);
        return res.json();
      })
      .then(data => {
        setUser(data);
        setLoading(false);
      })
      .catch(err => {
        if (err.name === 'AbortError') return; // cleanly ignored
        setError(err.message);
        setLoading(false);
      });

    // Cleanup aborts request if user searches again before fetch finishes
    return () => controller.abort();
  }, [username]);

  return (
    <div>
      <input value={input} onChange={e => setInput(e.target.value)} />
      <button onClick={() => setUsername(input.trim())}>Search</button>
      ...
    </div>
  );
}
// Race condition prevention with AbortController
1. User types "m"     -> effect enqueues fetch(/users/m)
2. User types "mu"    -> cleanup fires: controller.abort() cancels fetch(/users/m)
3. New effect starts  -> fetch(/users/mu) enqueued
4. Result arrives     -> Only the latest query completes and updates state!
// No out-of-order stale data overwrites the screen!
