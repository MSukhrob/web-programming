Homework Specification (Slide 39)

Build a real-time weather dashboard using the Open-Meteo API (no key needed). The component must fetch forecast data inside useEffect whenever coordinates change, manage loading and error states, and clean up in-flight requests with AbortController.

  • Open-Meteo Forecast API: Fetches temperature, wind speed, weather code, and 7-day daily forecast.
  • Geocoding Endpoint: Type a city name (e.g. Seoul, Tashkent, Tokyo) to look up coordinates automatically.
  • AbortController Cleanup: Cancels pending network requests when switching locations or unmounting.
  • Bonus: One-click navigator.geolocation auto-detection for current user coordinates.

// Geocoding & WMO Weather Code Interpreter
async function searchCity(cityName, signal) {
  const geoUrl = `https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(cityName)}&count=1&language=en&format=json`;
  const res = await fetch(geoUrl, { signal });
  const data = await res.json();
  if (!data.results || data.results.length === 0) {
    throw new Error(`City "${cityName}" not found.`);
  }
  const top = data.results[0];
  return { lat: top.latitude, lon: top.longitude, city: `${top.name}, ${top.country}` };
}

// WMO Standard Weather Code Descriptions & Icons
function getWeatherMeta(code) {
  if (code === 0) return { label: 'Clear Sky', icon: '☀️' };
  if (code === 1 || code === 2) return { label: 'Partly Cloudy', icon: '⛅' };
  if (code === 3) return { label: 'Overcast', icon: '☁️' };
  if ([45, 48].includes(code)) return { label: 'Foggy', icon: '🌫️' };
  if ([51, 53, 55, 61, 63, 65].includes(code)) return { label: 'Rainy', icon: '🌧️' };
  if ([71, 73, 75, 77, 85, 86].includes(code)) return { label: 'Snowy', icon: '❄️' };
  if ([95, 96, 99].includes(code)) return { label: 'Thunderstorm', icon: '⛈️' };
  return { label: 'Variable', icon: '🌤️' };
}
