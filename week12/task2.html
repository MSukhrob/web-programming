Challenge Specifications (Slide 33)

Build a Counter component with the following stateful capabilities:

  1. Display current count: Starts at 0.
  2. +1 and −1 buttons: Schedule state updates.
  3. Reset button: Returns the counter to 0 and logs reset in history.
  4. History: Stores and displays the last 5 count values: [0, 1, 2, 1, 0].
  5. Undo: Sets count to the previous value in history (history[history.length - 2]), then immutably removes the latest item from history using h.slice(0, -1).

// 07-CounterWithHistory.jsx
import { useState } from 'react';

export default function CounterWithHistory() {
  const [count, setCount]     = useState(0);
  const [history, setHistory] = useState([0]);

  function change(next) {
    setCount(next);
    // Keep last 5 count values
    setHistory(h => [...h, next].slice(-5));
  }

  function undo() {
    if (history.length < 2) return;
    const previous = history[history.length - 2];
    setCount(previous);
    setHistory(h => h.slice(0, -1));
  }

  return (
    <div className="counter-box">
      <h2>{count}</h2>
      <button onClick={() => change(count + 1)}>+1</button>
      <button onClick={() => change(count - 1)}>-1</button>
      <button onClick={() => change(0)}>Reset</button>
      <button onClick={undo} disabled={history.length < 2}>Undo</button>
      <p>History: [{history.join(', ')}]</p>
    </div>
  );
}
