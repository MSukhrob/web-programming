Homework Requirements (Slide 34)

Implement an interactive shopping cart application demonstrating complete React state management:

  • Product List (1): At least 5 items with name, formatted price (KRW), and image URL.
  • Add / Remove Buttons (2): Array of cart items stored in React state.
  • Cart Sidebar (3): Displays added items and computes the real-time total price directly from state.
  • Quantity Controls (4): + and − per item with immutable array mapping (prev.map() / prev.filter()).
  • ShoppingCart Component (5): Parent component composing the catalog and cart sidebar via lifting state up.
  • Bonus features: Promo code discount (try: WEBPRO10), live item badge, and order confirmation.

// ShoppingCart.jsx - Immutable array state management
const [cart, setCart] = useState([]);

// 1. Add item: Check if exists, increment quantity or append new object
const addToCart = (product) => {
  setCart(prev => {
    const existing = prev.find(item => item.id === product.id);
    if (existing) {
      return prev.map(item =>
        item.id === product.id ? { ...item, qty: item.qty + 1 } : item
      );
    }
    return [...prev, { ...product, qty: 1 }];
  });
};

// 2. Update Quantity: Immutably update or remove when qty reaches 0
const updateQty = (id, delta) => {
  setCart(prev =>
    prev
      .map(item => item.id === id ? { ...item, qty: item.qty + delta } : item)
      .filter(item => item.qty > 0)
  );
};

// 3. Remove Item completely
const removeFromCart = (id) => {
  setCart(prev => prev.filter(item => item.id !== id));
};

function ProductCard({ product, onAdd }) {
  return (
    <div className="shop-product-card">
      <img src={product.image} alt={product.name} className="product-img" />
      <div className="product-info">
        <h3>{product.name}</h3>
        <p className="product-price-tag">{product.price.toLocaleString('ko-KR')}원</p>
        <button className="btn" onClick={() => onAdd(product)}>
          + Add to Cart
        </button>
      </div>
    </div>
  );
}
// CartSidebar.jsx
function CartSidebar({ cart, onUpdateQty, onRemove, onClear }) {
  const subtotal = cart.reduce((sum, item) => sum + item.price * item.qty, 0);

  return (
    <aside className="cart-sidebar">
      <h2>Your Cart ({cart.reduce((s, i) => s + i.qty, 0)})</h2>
      {cart.map(item => (
        <div key={item.id} className="cart-item-row">
          <div>
            <div>{item.name}</div>
            <div>{(item.price * item.qty).toLocaleString()}원</div>
          </div>
          <div>
            <button onClick={() => onUpdateQty(item.id, -1)}>−</button>
            <span>{item.qty}</span>
            <button onClick={() => onUpdateQty(item.id, 1)}>+</button>
          </div>
        </div>
      ))}
      <h3>Total: {subtotal.toLocaleString()}원</h3>
    </aside>
  );
}
